server {
    listen 127.0.0.1:8081;
    server_name localhost;

    access_log /var/log/nginx/network-lab.access.log;
    error_log /var/log/nginx/network-lab.error.log;

    add_header X-Lab-Proxy nginx always;

    # NGINX handles this endpoint directly
    location = / {
        default_type text/plain;
        return 200 "NGINX is running\n";
    }

    # Do not expose the application's metrics endpoint
    # through this demo API route
    location = /api/metrics {
        return 404;
    }

    # Forward API requests to Go
    location /api/ {
        proxy_pass http://127.0.0.1:8080/;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Reject other paths
    location / {
        return 404;
    }
}
