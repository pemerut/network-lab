server {
    listen 127.0.0.1:8081;
    server_name localhost;

    access_log /var/log/nginx/network-lab.access.log lab_upstream;
    error_log /var/log/nginx/network-lab.error.log;

    add_header X-Lab-Proxy nginx always;
    add_header X-Upstream $upstream_addr always;

    # NGINX handles this endpoint directly
    location = / {
        default_type text/plain;
        return 200 "NGINX is running\n";
    }

    # Do not expose the application's metrics endpoint
    # through this demo API route
    location = /api/metrics {
        return 404;
    }

    # Forward API requests to Go
    location /api/ {
        proxy_pass http://go_backend/;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout 2s;
        proxy_read_timeout 10s;

        proxy_next_upstream error timeout;
        proxy_next_upstream_tries 2;
    }

    # Reject other paths
    location / {
        return 404;
    }
}
