sudo install -d -m 0700 /etc/nginx/lab-tls

sudo openssl req -x509 \
  -newkey rsa:2048 \
  -nodes \
  -days 30 \
  -keyout /etc/nginx/lab-tls/key.pem \
  -out /etc/nginx/lab-tls/cert.pem \
  -subj "/CN=localhost" \
  -addext "subjectAltName=DNS:localhost,IP:127.0.0.1"

  sudo chmod 600 /etc/nginx/lab-tls/key.pem
sudo chmod 644 /etc/nginx/lab-tls/cert.pem
